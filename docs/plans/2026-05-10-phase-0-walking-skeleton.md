These are NOT detailed plans. Each phase gets its own detailed plan written when prior phase completes. These outlines exist so context is visible while writing Phase 0.

### Phase 1 — Disk storage (~6-8 wks)
- Replace Phase 0's in-memory Store impl with a CoW B+-tree on the BLOCK layer.
- Add `Unix_file` BLOCK backend (POSIX file).
- Add page format (header pages, leaf, branch, freelist).
- Add atomic-commit protocol (alternating headers, two fsyncs).
- Add page cache + freelist + page allocator.
- Add CRC32 on header pages.
- Crash-recovery on open.
- Extend SQL: `ORDER BY`, `LIMIT`/`OFFSET`, `CREATE INDEX`, `IndexLookup`/`IndexRangeScan` operators.

### Phase 3 — Transactions/MVCC (~6 wks)

### Phase 4 — MirageOS (~4 wks)

```ocaml
let page_size = 4096

type error =
  | Out_of_bounds of { page_id : int64; n_pages : int64 }

let pp_error fmt = function
  | Out_of_bounds { page_id; n_pages } ->
    Format.fprintf fmt
      "Out_of_bounds page_id=%Ld n_pages=%Ld" page_id n_pages

type t = {
  mutable pages : Bytes.t array;  (* page_id → page bytes *)
  mutable n_pages : int64;
}

let create ~n_pages =
  let n = Int64.to_int n_pages in
  let pages = Array.init n (fun _ -> Bytes.make page_size '\x00') in
  { pages; n_pages }

let n_pages t = t.n_pages

let in_bounds t page_id =
  Int64.compare page_id 0L >= 0
  && Int64.compare page_id t.n_pages < 0

let read_page t ~page_id buf =
  if not (in_bounds t page_id) then
    Lwt.return (Error (Out_of_bounds { page_id; n_pages = t.n_pages }))
  else begin
    let src = t.pages.(Int64.to_int page_id) in
    Cstruct.blit_from_bytes src 0 buf 0 page_size;
    Lwt.return (Ok ())
  end

let write_page t ~page_id buf =
  if not (in_bounds t page_id) then
    Lwt.return (Error (Out_of_bounds { page_id; n_pages = t.n_pages }))
  else begin
    let dst = t.pages.(Int64.to_int page_id) in
    Cstruct.blit_to_bytes buf 0 dst 0 page_size;
    Lwt.return (Ok ())
  end

let sync _ = Lwt.return (Ok ())

let resize t ~n_pages =
  let n = Int64.to_int n_pages in
  let cur = Array.length t.pages in
  if n > cur then begin
    let extra = Array.init (n - cur) (fun _ -> Bytes.make page_size '\x00') in
    t.pages <- Array.append t.pages extra
  end else if n < cur then
    t.pages <- Array.sub t.pages 0 n;
  t.n_pages <- n_pages;
  Lwt.return (Ok ())
```

```ocaml
    Uses zigzag for int64 (so negatives are short for small magnitudes). *)

val encode_int64 : Buffer.t -> int64 -> unit
val decode_int64 : Bytes.t -> int -> int64 * int
```

- [ ] **Step 6: Write `lib/encoding/varint.ml`**

```ocaml
let encode_uint64 buf n =
  let n = ref n in
  while Int64.compare (Int64.shift_right_logical !n 7) 0L <> 0 do
    let b = Int64.to_int (Int64.logor (Int64.logand !n 0x7FL) 0x80L) in
    Buffer.add_char buf (Char.chr (b land 0xFF));
    n := Int64.shift_right_logical !n 7
  done;
  Buffer.add_char buf (Char.chr (Int64.to_int !n land 0x7F))

let decode_uint64 buf off =
  let result = ref 0L in
  let shift = ref 0 in
  let off = ref off in
  let cont = ref true in
  while !cont do
    let b = Char.code (Bytes.get buf !off) in
    incr off;
    let payload = Int64.of_int (b land 0x7F) in
    result := Int64.logor !result (Int64.shift_left payload !shift);
    shift := !shift + 7;
    if b land 0x80 = 0 then cont := false
  done;
  !result, !off

let zigzag_encode n =
  Int64.logxor (Int64.shift_left n 1) (Int64.shift_right n 63)

let zigzag_decode n =
  Int64.logxor (Int64.shift_right_logical n 1) (Int64.neg (Int64.logand n 1L))

let encode_int64 buf n = encode_uint64 buf (zigzag_encode n)
let decode_int64 buf off =
  let u, off = decode_uint64 buf off in
  zigzag_decode u, off
```

```ocaml
(* 8-byte big-endian, sign bit flipped. This places negatives below
   positives under bytewise comparison. *)

let encode n =
  let n = Int64.logxor n 0x8000_0000_0000_0000L in (* flip sign bit *)
  let b = Bytes.create 8 in
  for i = 0 to 7 do
    let byte = Int64.to_int (Int64.shift_right_logical n ((7 - i) * 8)) land 0xFF in
    Bytes.set_uint8 b i byte
  done;
  b

let decode b =
  let n = ref 0L in
  for i = 0 to 7 do
    let byte = Bytes.get_uint8 b i in
    n := Int64.logor !n (Int64.shift_left (Int64.of_int byte) ((7 - i) * 8))
  done;
  Int64.logxor !n 0x8000_0000_0000_0000L (* unflip sign bit *)
```

```ocaml
type ty =
  | Integer
  | Text

type column = { name : string; ty : ty }
type schema = column list

type value =
  | V_int of int64
  | V_text of string
  | V_null

type t = value array

let value_equal a b = match a, b with
  | V_int x, V_int y -> Int64.equal x y
  | V_text x, V_text y -> String.equal x y
  | V_null, V_null -> true
  | _ -> false

let equal a b =
  Array.length a = Array.length b
  && Array.for_all2 value_equal a b

let encode schema row =
  let n = List.length schema in
  if Array.length row <> n then
    invalid_arg "Row.encode: row arity mismatch";
  let buf = Buffer.create 32 in
  (* n_cols varint *)
  Varint.encode_uint64 buf (Int64.of_int n);
  (* null bitmap *)
  let bitmap_bytes = (n + 7) / 8 in
  let bitmap = Bytes.make bitmap_bytes '\x00' in
  Array.iteri (fun i v ->
    if v = V_null then begin
      let byte = i / 8 and bit = i mod 8 in
      let cur = Bytes.get_uint8 bitmap byte in
      Bytes.set_uint8 bitmap byte (cur lor (1 lsl bit))
    end
  ) row;
  Buffer.add_bytes buf bitmap;
  (* values *)
  List.iteri (fun i col ->
    match row.(i) with
    | V_null -> ()  (* skip *)
    | V_int n ->
      (match col.ty with
       | Integer -> Varint.encode_int64 buf n
       | Text -> invalid_arg "Row.encode: int in text column")
    | V_text s ->
      (match col.ty with
       | Text ->
         Varint.encode_uint64 buf (Int64.of_int (String.length s));
         Buffer.add_string buf s
       | Integer -> invalid_arg "Row.encode: text in int column")
  ) schema;
  Buffer.to_bytes buf

let decode schema bytes =
  let n = List.length schema in
  let n', off = Varint.decode_uint64 bytes 0 in
  if Int64.to_int n' <> n then
    invalid_arg "Row.decode: arity mismatch";
  let bitmap_bytes = (n + 7) / 8 in
  let bitmap = Bytes.sub bytes off bitmap_bytes in
  let off = ref (off + bitmap_bytes) in
  let result = Array.make n V_null in
  List.iteri (fun i col ->
    let byte = i / 8 and bit = i mod 8 in
    let is_null = (Bytes.get_uint8 bitmap byte lsr bit) land 1 = 1 in
    if is_null then result.(i) <- V_null
    else match col.ty with
      | Integer ->
        let n, off' = Varint.decode_int64 bytes !off in
        result.(i) <- V_int n;
        off := off'
      | Text ->
        let len, off' = Varint.decode_uint64 bytes !off in
        let len = Int64.to_int len in
        let s = Bytes.sub_string bytes off' len in
        result.(i) <- V_text s;
        off := off' + len
  ) schema;
  result
```

```ocaml
(* Phase 0 in-memory store. Phase 1 will replace this with CoW B+-tree
   on BLOCK behind the same .mli. *)

module BytesMap = Map.Make (Bytes)

type ro
type rw

type tree_id = int

type t = {
  trees : (tree_id, Bytes.t BytesMap.t ref) Hashtbl.t;
  mutable closed : bool;
}

(* Phase 0: txn carries no per-txn state. rw_begin is mutually exclusive
   via a Lwt_mutex; commit/rollback are effectively no-ops because mutations
   apply directly. Phase 3 introduces real snapshot isolation. *)

let rw_mutex = Lwt_mutex.create ()

type 'a txn =
  | Ro : t -> ro txn
  | Rw : { store : t; mutex_token : unit Lwt.t Lazy.t } -> rw txn

let create () = { trees = Hashtbl.create 16; closed = false }

let close t =
  t.closed <- true;
  Lwt.return_unit

let ro_begin t =
  if t.closed then Lwt.fail_with "store closed"
  else Lwt.return (Ro t)

let rw_begin t =
  if t.closed then Lwt.fail_with "store closed"
  else
    let%lwt () = Lwt_mutex.lock rw_mutex in
    Lwt.return (Rw { store = t; mutex_token = lazy (Lwt.return_unit) })

let commit (Rw _ : rw txn) =
  Lwt_mutex.unlock rw_mutex;
  Lwt.return_unit

let rollback (Rw _ : rw txn) =
  (* Phase 0: in-memory mutations cannot be rolled back. Documented
     limitation; Phase 3 introduces real rollback. *)
  Lwt_mutex.unlock rw_mutex;
  Lwt.return_unit

let ro_end (Ro _ : ro txn) = Lwt.return_unit

let store_of : type a. a txn -> t = function
  | Ro s -> s
  | Rw { store; _ } -> store

let tree (s : t) tid =
  match Hashtbl.find_opt s.trees tid with
  | Some r -> r
  | None ->
    let r = ref BytesMap.empty in
    Hashtbl.add s.trees tid r;
    r

let get : type a. a txn -> tree_id -> bytes -> bytes option Lwt.t =
  fun tx tid key ->
    let s = store_of tx in
    let r = tree s tid in
    Lwt.return (BytesMap.find_opt key !r)

let put (Rw { store; _ } : rw txn) tid key value =
  let r = tree store tid in
  r := BytesMap.add key value !r;
  Lwt.return_unit

let del (Rw { store; _ } : rw txn) tid key =
  let r = tree store tid in
  r := BytesMap.remove key !r;
  Lwt.return_unit

type seek_result =
  | Found of bytes
  | Not_found of [`Greater of bytes | `End]

(** Cursor semantics:
    - cursor_first/cursor_seek position the cursor.
    - cursor_next consumes one entry and returns it; on first call after
      open or after cursor_first, returns the first entry. *)
type cursor = {
  all : (bytes * bytes) list;       (* immutable snapshot *)
  mutable remaining : (bytes * bytes) list;
}

let cursor_open : type a. a txn -> tree_id -> cursor Lwt.t =
  fun tx tid ->
    let s = store_of tx in
    let r = tree s tid in
    let entries = BytesMap.bindings !r in
    Lwt.return { all = entries; remaining = entries }

let cursor_close _ = ()

let cursor_first c =
  c.remaining <- c.all;
  match c.all with
  | [] -> Not_found `End
  | (k, _) :: _ -> Found k

let cursor_seek c key =
  let rec find = function
    | [] -> c.remaining <- []; Not_found `End
    | ((k, _) :: _) as cur ->
      let cmp = Bytes.compare k key in
      if cmp >= 0 then begin
        c.remaining <- cur;
        if cmp = 0 then Found k else Not_found (`Greater k)
      end else find (List.tl cur)
  in
  c.remaining <- c.all;
  find c.all

let cursor_next c =
  match c.remaining with
  | [] -> None
  | (k, v) :: rest ->
    c.remaining <- rest;
    Some (k, v)

let cursor_value c =
  match c.remaining with
  | [] -> None
  | (_, v) :: _ -> Some v
```

```ocaml
        | None -> ()
        | Some (k, v) when Bytes.length k >= Bytes.length prefix
                        && Bytes.sub k 0 (Bytes.length prefix) = prefix ->
          cols := decode_column v :: !cols;
          walk2 ()
        | _ -> ()
      in
      walk2 ();
      S.cursor_close cur2;
      Hashtbl.add tbl name { name; tree_id = tid;
                             columns = List.rev !cols; next_rowid };
      walk ()
  in
  let%lwt () = walk () in
  S.cursor_close cur;
  let%lwt () = S.ro_end tx in
  Lwt.return tbl

let open_ store =
  let%lwt cache = load_meta_table_data store in
  Lwt.return { store; cache }

let next_user_tid t : int Lwt.t =
  let%lwt tx = S.ro_begin t.store in
  let%lwt v = S.get tx sys_meta_tid next_user_tid_key in
  let%lwt () = S.ro_end tx in
  match v with
  | Some b ->
    let n, _ = Varint.decode_uint64 b 0 in
    Lwt.return (Int64.to_int n)
  | None -> Lwt.return next_user_tid_init

let bump_user_tid t : unit Lwt.t =
  let%lwt cur = next_user_tid t in
  let next = cur + 1 in
  let%lwt tx = S.rw_begin t.store in
  let buf = Buffer.create 8 in
  Varint.encode_uint64 buf (Int64.of_int next);
  let%lwt () =
    S.put tx sys_meta_tid next_user_tid_key (Buffer.to_bytes buf) in
  S.commit tx

let create_table t ~name ~columns =
  if Hashtbl.mem t.cache name then
    failwith (Printf.sprintf "table %s already exists" name);
  let%lwt tid = next_user_tid t in
  let%lwt () = bump_user_tid t in
  let m = { name; tree_id = tid; columns; next_rowid = 1L } in
  let%lwt tx = S.rw_begin t.store in
  let%lwt () =
    S.put tx sys_tables_tid (Bytes.of_string name) (encode_table_value m) in
  let%lwt () =
    Lwt_list.iteri_s (fun i col ->
      let key = column_key name i in
      let v = encode_column col in
      S.put tx sys_columns_tid key v
    ) columns in
  let%lwt () = S.commit tx in
  Hashtbl.add t.cache name m;
  Lwt.return tid

let find_table t ~name =
  Lwt.return (Hashtbl.find_opt t.cache name)

let list_tables t =
  Lwt.return (Hashtbl.fold (fun _ v acc -> v :: acc) t.cache [])

let next_rowid t ~name =
  match Hashtbl.find_opt t.cache name with
  | None -> failwith (Printf.sprintf "no table %s" name)
  | Some m ->
    let id = m.next_rowid in
    let m' = { m with next_rowid = Int64.add id 1L } in
    Hashtbl.replace t.cache name m';
    let%lwt tx = S.rw_begin t.store in
    let%lwt () =
      S.put tx sys_tables_tid
        (Bytes.of_string name) (encode_table_value m') in
    let%lwt () = S.commit tx in
    Lwt.return id
```

- [ ] **Step 3: Write `lib/sql/sema.ml`**

```ocaml
open Lwt.Syntax
module Row = Sqlocaml_encoding.Row
module Cat = Sqlocaml_catalog.Catalog

type bound_expr =
  | BE_lit of Ast.literal
  | BE_col of int
  | BE_eq of bound_expr * bound_expr

type bound_stmt =
  | BS_create_table of { name : string; columns : Row.column list }
  | BS_insert of {
      table_meta : Cat.table_meta;
      ordinals : int list;
      values : Ast.literal list;
    }
  | BS_select of {
      table_meta : Cat.table_meta;
      proj : int list;
      where : bound_expr option;
    }

type error =
  | Unknown_table of string
  | Unknown_column of { table : string; column : string }
  | Type_mismatch of { expected : Row.ty; got : Row.ty }
  | Arity_mismatch of { expected : int; got : int }
  | Already_exists of string

let ty_of_ast = function
  | Ast.Ty_int -> Row.Integer
  | Ast.Ty_text -> Row.Text

let column_index columns name =
  let rec idx i = function
    | [] -> None
    | (c : Row.column) :: _ when c.name = name -> Some i
    | _ :: rest -> idx (i + 1) rest
  in
  idx 0 columns

let lit_ty = function
  | Ast.L_int _ -> Some Row.Integer
  | Ast.L_text _ -> Some Row.Text
  | Ast.L_null -> None

let rec bind_expr ~table_meta expr =
  let columns = table_meta.Cat.columns in
  match expr with
  | Ast.E_lit l -> Ok (BE_lit l)
  | Ast.E_col name ->
    begin match column_index columns name with
    | None -> Error (Unknown_column { table = table_meta.name; column = name })
    | Some i -> Ok (BE_col i)
    end
  | Ast.E_eq (a, b) ->
    begin match bind_expr ~table_meta a with
    | Error e -> Error e
    | Ok ba ->
      begin match bind_expr ~table_meta b with
      | Error e -> Error e
      | Ok bb -> Ok (BE_eq (ba, bb))
      end
    end

let resolve_insert_columns meta target_cols values =
  let rec go cols vals acc_ords =
    match cols, vals with
    | [], [] -> Ok (List.rev acc_ords)
    | name :: cs, v :: vs ->
      begin match column_index meta.Cat.columns name with
      | None -> Error (Unknown_column { table = meta.name; column = name })
      | Some i ->
        let col = List.nth meta.columns i in
        begin match lit_ty v with
        | None -> go cs vs (i :: acc_ords)
        | Some t when t = col.ty -> go cs vs (i :: acc_ords)
        | Some t -> Error (Type_mismatch { expected = col.ty; got = t })
        end
      end
    | _ -> Error (Arity_mismatch {
        expected = List.length target_cols;
        got = List.length values
      })
  in
  go target_cols values []

let resolve_select_proj meta = function
  | `All -> Ok (List.mapi (fun i _ -> i) meta.Cat.columns)
  | `Cols names ->
    let rec go = function
      | [] -> Ok []
      | n :: rest ->
        begin match column_index meta.columns n with
        | None -> Error (Unknown_column { table = meta.name; column = n })
        | Some i ->
          begin match go rest with
          | Error e -> Error e
          | Ok r -> Ok (i :: r)
          end
        end
    in
    go names

let bind_create_table cat ~name ~columns =
  let* m = Cat.find_table cat ~name in
  match m with
  | Some _ -> Lwt.return (Error (Already_exists name))
  | None ->
    let cols = List.map (fun (c : Ast.column_def) ->
      Row.{ name = c.name; ty = ty_of_ast c.ty }
    ) columns in
    Lwt.return (Ok (BS_create_table { name; columns = cols }))

let bind_insert cat ~table ~columns ~values =
  let* m = Cat.find_table cat ~name:table in
  match m with
  | None -> Lwt.return (Error (Unknown_table table))
  | Some meta ->
    let target_cols =
      if columns = [] then List.map (fun c -> c.Row.name) meta.columns
      else columns
    in
    if List.length target_cols <> List.length values then
      Lwt.return (Error (Arity_mismatch {
        expected = List.length target_cols;
        got = List.length values
      }))
    else
      begin match resolve_insert_columns meta target_cols values with
      | Error e -> Lwt.return (Error e)
      | Ok ordinals ->
        Lwt.return (Ok (BS_insert { table_meta = meta; ordinals; values }))
      end

let bind_select cat ~proj ~table ~where =
  let* m = Cat.find_table cat ~name:table in
  match m with
  | None -> Lwt.return (Error (Unknown_table table))
  | Some meta ->
    begin match resolve_select_proj meta proj with
    | Error e -> Lwt.return (Error e)
    | Ok ords ->
      let bound_where =
        match where with
        | None -> Ok None
        | Some e ->
          begin match bind_expr ~table_meta:meta e with
          | Error e -> Error e
          | Ok be -> Ok (Some be)
          end
      in
      begin match bound_where with
      | Error e -> Lwt.return (Error e)
      | Ok w -> Lwt.return (Ok (BS_select {
          table_meta = meta; proj = ords; where = w
```

```ocaml
let rec eval_expr (row : Row.t) (e : Plan.expr) : Row.value =
  match e with
  | Plan.P_lit l -> lit_to_value l
  | Plan.P_col i -> row.(i)
  | Plan.P_eq (a, b) ->
    let va = eval_expr row a and vb = eval_expr row b in
    let eq = match va, vb with
      | Row.V_int x, Row.V_int y -> Int64.equal x y
      | Row.V_text x, Row.V_text y -> String.equal x y
      | Row.V_null, _ | _, Row.V_null -> false
      | _ -> false
    in
    if eq then Row.V_int 1L else Row.V_int 0L

let value_truthy = function
  | Row.V_int 0L | Row.V_null -> false
  | _ -> true

let project_row (ords : int list) (row : Row.t) : Row.t =
  Array.of_list (List.map (fun i -> row.(i)) ords)

(* Build an Lwt_stream over a plan tree *)
let rec to_stream (store : S.t) (op : Plan.op)
    : Row.t Lwt_stream.t Lwt.t =
  match op with
  | Plan.Op_seq_scan { table_meta } ->
    let* tx = S.ro_begin store in
    let* cur = S.cursor_open tx table_meta.tree_id in
    let _ = S.cursor_first cur in
    let stream =
      Lwt_stream.from (fun () ->
        match S.cursor_next cur with
        | None ->
          S.cursor_close cur;
          let%lwt () = S.ro_end tx in
          Lwt.return None
        | Some (_, vbytes) ->
          let row = Row.decode table_meta.columns vbytes in
          Lwt.return (Some row))
    in
    Lwt.return stream
  | Plan.Op_filter { pred; child } ->
    let* inner = to_stream store child in
    Lwt.return (Lwt_stream.filter (fun row -> value_truthy (eval_expr row pred)) inner)
  | Plan.Op_project { ordinals; child } ->
    let* inner = to_stream store child in
    Lwt.return (Lwt_stream.map (project_row ordinals) inner)
  | Plan.Op_create_table _ | Plan.Op_insert _ ->
    failwith "Use Exec.execute for write ops"

let query store _cat op = to_stream store op
```

```ocaml
open Lwt.Syntax
module S = Sqlocaml_store.Store
module Cat = Sqlocaml_catalog.Catalog
module Sql = Sqlocaml_sql

type t = {
  store : S.t;
  catalog : Cat.t;
}

type value = Sqlocaml_encoding.Row.value =
  | V_int of int64
  | V_text of string
  | V_null

type row = Sqlocaml_encoding.Row.t

type error =
  | Parse of string
  | Sema of Sql.Sema.error
  | Storage_io of string

let open_in_memory () =
  let store = S.create () in
  let* catalog = Cat.open_ store in
  Lwt.return { store; catalog }

let close t =
  let* () = S.close t.store in
  Lwt.return_unit

let parse sql =
  try
    let lexbuf = Lexing.from_string sql in
    Ok (Sql.Parser.stmt_eof Sql.Lexer.token lexbuf)
  with
  | Failure msg -> Error (Parse msg)
  | Sql.Parser.Error -> Error (Parse "syntax error")

let prepare t sql =
  match parse sql with
  | Error e -> Lwt.return (Error e)
  | Ok ast ->
    let* bound = Sql.Sema.bind t.catalog ast in
    match bound with
    | Error e -> Lwt.return (Error (Sema e))
    | Ok b -> Lwt.return (Ok (Sql.Planner.plan b))

let execute t sql =
  let* op = prepare t sql in
  match op with
  | Error e -> Lwt.return (Error e)
  | Ok op ->
    let* () = Sql.Exec.execute t.store t.catalog op in
    Lwt.return (Ok ())

let query t sql =
  let* op = prepare t sql in
  match op with
  | Error e -> Lwt.return (Error e)
  | Ok op ->
    let* stream = Sql.Exec.query t.store t.catalog op in
    Lwt.return (Ok stream)
```
