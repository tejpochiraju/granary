### Step 2: Nested trigger firing

Current state: `lib/db/db.ml:350` (`fire_trigger_stmt`) compiles each trigger-body statement and calls `Sql.Exec.execute` directly, bypassing the `before_hook` / `after_hook` arguments. Hooks are only attached at the top-level `Db.execute` call site. Result: a DML statement inside a trigger body does not fire any matching triggers.

We need fire_trigger_stmt to install hooks for the nested DML, with recursion-depth tracking to prevent infinite loops.


- [ ] **Step 2.2: Refactor `fire_trigger_stmt` to install nested hooks**

Replace the body of `fire_trigger_stmt`:

```ocaml
let max_trigger_depth = 32

let rec fire_trigger_stmt t stmt =
  if t.trigger_depth >= max_trigger_depth then
    Lwt.fail_with (Printf.sprintf
      "trigger recursion limit (%d) exceeded" max_trigger_depth)
  else begin
    t.trigger_depth <- t.trigger_depth + 1;
    let finally () = t.trigger_depth <- t.trigger_depth - 1; Lwt.return_unit in
    Lwt.finalize (fun () ->
      let* bound = Sql.Sema.bind ~views:t.views t.catalog stmt in
      match bound with
      | Error e -> Lwt.fail_with (Format.asprintf "trigger sema: %a" Sql.Sema.pp_error e)
      | Ok b ->
        let op = Sql.Planner.plan ~cat:t.catalog b in
        let mode = match t.explicit_txn with
          | None    -> Sql.Exec.Auto
          | Some tx -> Sql.Exec.In_txn tx
        in
        (* Build hooks for the nested op. We look up the target table from the op. *)
        let table_name_opt = match op with
          | Sql.Plan.Op_insert { table_meta; _ }
          | Sql.Plan.Op_insert_select { table_meta; _ }
          | Sql.Plan.Op_update { table_meta; _ }
          | Sql.Plan.Op_delete { table_meta; _ } -> Some table_meta.Sql.Catalog.name
          | _ -> None
        in
        let table_meta_opt =
          Option.bind table_name_opt (fun n ->
            Sql.Catalog.find_table_cached t.catalog ~name:n)
        in
        let event_opt = match op with
          | Sql.Plan.Op_insert _ | Sql.Plan.Op_insert_select _ -> Some `Insert
          | Sql.Plan.Op_update _ -> Some `Update
          | Sql.Plan.Op_delete _ -> Some `Delete
          | _ -> None
        in
        let before_hook, after_hook =
          match table_meta_opt, event_opt with
          | Some tm, Some ev ->
            make_trigger_hook t tm ~timing:`Before ~event:ev,
            make_trigger_hook t tm ~timing:`After  ~event:ev
          | _ -> None, None
        in
        Sql.Exec.execute ?before_hook ?after_hook ~mode ~clock:t.clock t.store t.catalog op
    ) finally
  end
```

The previous comment "triggers fired here do NOT recursively fire further triggers" should be deleted.

Note: the `make_trigger_hook` and `fire_trigger_stmt` are now mutually recursive through hook capture — declare with `let rec ... and ...` OR (simpler) restructure so `make_trigger_hook` references `fire_trigger_stmt` via the same scope. They are already in the same module and the existing `make_trigger_hook` already calls `fire_trigger_stmt`. The forward reference works if both use `let rec ... and ...`. If they don't compile that way, use a `ref` indirection.

Pragmatically: add `let rec fire_trigger_stmt ... = ... and make_trigger_hook ... = ...`. Combine them into one `let rec ... and ...` chain.

### Step 3: INSTEAD OF UPDATE OLD row population

- [ ] **Step 3.2: Add OLD-row population**

Modify the arm to read matching view rows first. Pattern:

```ocaml
| Sql.Ast.S_update { table = _; assignments; where; _ } ->
  let matching = find_instead_of `Update in
  if matching = [] then
    Lwt.return (Error (Sema (Sql.Sema.Unsupported
      (Printf.sprintf "view '%s' is not directly modifiable (no INSTEAD OF UPDATE trigger)"
         view_name))))
  else begin
    let view_query = Hashtbl.find_opt t.views view_name in
    let assign_cols  = List.map fst assignments in
    let assign_exprs = List.map snd assignments in
    (* Compose: SELECT * FROM view_name WHERE <where> *)
    let* old_rows =
      match view_query with
      | None -> Lwt.return []
      | Some _ ->
        let select_sql =
          match where with
          | None -> Printf.sprintf "SELECT * FROM %s" view_name
          | Some w -> Printf.sprintf "SELECT * FROM %s WHERE %s"
                        view_name (Sql.Ast.expr_to_sql w)
        in
        let* result = query t select_sql in
        (match result with
         | Error _ -> Lwt.return []
         | Ok stream -> Lwt_stream.to_list stream)
    in
    let old_col_names =
      match view_query with
      | Some vq -> view_col_names vq
      | None -> []
    in
    let old_schema = make_col_schema old_col_names in
    let new_vals = List.map eval_insert_ast_value assign_exprs in
    let new_schema = make_col_schema assign_cols in
    let new_row = Some (Array.of_list new_vals) in
    let process_one_old_row old_row_opt =
      Lwt_list.iter_s (fun m ->
        let substituted_body = List.map (fun stmt ->
          (* substitute NEW first with new_schema, then OLD with old_schema *)
          let s1 = subst_new_old ~schema:new_schema ~new_row ~old_row:None stmt in
          subst_new_old ~schema:old_schema ~new_row:None ~old_row:old_row_opt s1
        ) m.trig_body in
        Lwt_list.iter_s (fire_trigger_stmt t) substituted_body
      ) matching
    in
    let* () =
      if old_rows = [] then process_one_old_row None
      else Lwt_list.iter_s (fun r -> process_one_old_row (Some r)) old_rows
    in
    Lwt.return (Ok ())
  end
```

Key details:
- `subst_new_old` is called twice with disjoint schemas: once for NEW, once for OLD. This works because `subst_new_old` only replaces matching references and leaves others untouched.
- `query` is the existing `Db.query` function — it already routes through `execute_instead_of`'s sibling path. Use the actual symbol name.
- If `view_query` does not match the view-col-name pattern, OLD is empty and the trigger fires once with no OLD substitution (preserves Phase 32 behavior).

### Step 4: REPLACE / UPSERT-update trigger firing

Currently `OR REPLACE` (in `execute_insert` ~line 1880-1899) deletes conflicting rows directly via `S.del` without firing any DELETE trigger. The UPSERT `DO UPDATE` branch updates rows without firing UPDATE triggers either. We need both to fire.

The cleanest approach: rather than thread `before_hook` / `after_hook` through `execute_insert`, look up the trigger hooks at the call site and invoke them manually around the delete/update.

- [ ] **Step 4.1: Identify entry points**

```bash

grep -n "REPLACE: delete all conflicting" lib/sql/exec.ml
```

The REPLACE deletion loop is in `execute_insert` itself. Hooks are not directly in scope. Solution: pass two extra optional callbacks into `execute_insert`:

```ocaml
?on_replace_delete : (old_row:Row.t -> unit Lwt.t) option
?on_upsert_update  : (old_row:Row.t -> new_row:Row.t -> unit Lwt.t) option
```

- [ ] **Step 4.2: Plumb the new callbacks into `execute_insert`**

Add the parameters to the function signature. Inside the REPLACE block, after decoding `old_row`, call:


Split into two functions:

```ocaml
let compute_stored_generated_cols clock params meta row =
  List.iteri (fun i (col : Row.column) ->
    match col.Row.generated_as with
    | None -> ()
    | Some (sql, true) ->  (* STORED only *)
      let plan_e = compile_generated_expr meta.Cat.name i meta.Cat.columns sql in
      row.(i) <- eval_expr clock params row plan_e
    | Some (_, false) ->

      row.(i) <- Row.V_null

let compute_virtual_generated_cols clock params meta row =
  List.iteri (fun i (col : Row.column) ->
    match col.Row.generated_as with
    | Some (sql, false) ->  (* VIRTUAL only *)
      let plan_e = compile_generated_expr meta.Cat.name i meta.Cat.columns sql in
      row.(i) <- eval_expr clock params row plan_e
    | _ -> ()
  ) meta.Cat.columns
```
