# Phase 38 — Shared-read / exclusive-write WAL concurrency (#149)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Multiple concurrent reader fibers run snapshot-isolated against a single writer fiber on the WAL backend; SELECT throughput under contention rises ≥3× over the current single-`rw_mutex` serialisation.

**Architecture:** Replace `t.rw_mutex` (one `Lwt_mutex.t` serialising every txn) with a shared/exclusive primitive. Each RO snapshot captures the WAL `committed_frames` count at open time and only ever sees frames ≤ that bound. The WAL index becomes per-page frame history (newest-first list) so `find_page_at ~max_frame` resolves the snapshot's view in O(history-depth-per-page). The pager's read cache is re-keyed by `(page_id, version)` where version is the WAL frame index that produced the bytes (or `-1` for main DB) — making every cache entry content-immutable and thus safe across concurrent readers. Inline autocheckpoint is replaced by a background fiber gated on minimum active reader snapshot.

**Tech Stack:** OCaml + Lwt; modules touched: `lib/storage/wal.ml{,i}`, `lib/storage/pager.ml{,i}`, `lib/store/store.ml{,i}`; new module `lib/store/rwlock.ml{,i}`. Tests via Alcotest + Lwt under podman dune.

```ocaml
(* Snapshot-aware lookup: return the newest frame_idx for [pid] such
   that [idx < max_frame]. Note the strict "<": [max_frame] is the
   reader's [committed_frames] snapshot, which counts how many frames
   are visible — frame indices 0..max_frame-1.  Walks the list (newest
   first) so a 1-deep page returns in O(1); deeper histories cost
   proportionally to writes-per-page since checkpoint. *)
let find_page_at t pid ~max_frame =
  match Hashtbl.find_opt t.index pid with
  | None -> None
  | Some lst ->
    let rec scan = function
      | [] -> None
      | idx :: rest ->
        if idx < max_frame then Some idx
        else scan rest
    in
    scan lst
```

## Task 2 — Pager: versioned cache, snapshot-aware `read`, writer stops touching shared cache

**Why:** The shared `Pager.cache` is keyed by `page_id` only. Under concurrent readers + a writer, a writer's own `Pager.write` would inject uncommitted bytes into the cache, and a stale entry from frame N+1 would shadow a reader at snapshot N. Versioning the cache key by frame-idx makes every cache entry content-immutable; the writer keeps its own dirty hashtable as the single source of "what I'm writing".

**Files:**
- Modify: `lib/storage/pager.ml`
- Modify: `lib/storage/pager.mli`

```ocaml
let maybe_evict t =
  let cache_size = Hashtbl.length t.cache in
  if cache_size < cache_capacity then ()
  else begin
    let evicted = ref false in
    let temp = Queue.create () in
    while not !evicted && not (Queue.is_empty t.fifo) do
      let key = Queue.pop t.fifo in
      if Hashtbl.mem t.dirty (fst key) then
        Queue.push key temp
      else begin
        Hashtbl.remove t.cache key;
        evicted := true;
        Queue.iter (fun p -> Queue.push p t.fifo) temp;
        Queue.clear temp
      end
    done;
    if not !evicted then
      Queue.iter (fun p -> Queue.push p t.fifo) temp
  end

let cache_add t key buf =
  let already_cached = Hashtbl.mem t.cache key in
  maybe_evict t;
  Hashtbl.replace t.cache key buf;
  if not already_cached then
    Queue.push key t.fifo
```

### Step 2.3 — Add `?snapshot_frames` to `Pager.read`

- [ ] **Replace `lib/storage/pager.ml:114-162` (the current `read`)** with the snapshot-aware version:

```ocaml
(* read:
   - If [snapshot_frames] is [None] the caller is the writer or a non-WAL
     reader; resolves "latest committed" via [wal_find_page] and falls
     back to main DB.  Also honours [dirty] so the writer sees its own
     uncommitted edits.
   - If [snapshot_frames = Some n] the caller is an RO snapshot bounded
     to frames < n; ignores [dirty] (the writer's edits are invisible to
     a concurrent reader) and resolves via [wal_find_page_at]. *)
let read ?snapshot_frames t page_id =
  let open Lwt.Syntax in
  match snapshot_frames with
  | None ->
    (* Writer / non-snapshot path: dirty wins. *)
    (match Hashtbl.find_opt t.dirty page_id with
     | Some buf -> Lwt.return_ok (cstruct_dup buf)
     | None ->
       let resolve_via_wal () =
         match t.wal with
         | None -> Lwt.return_ok None
         | Some cb ->
           match cb.wal_find_page page_id with
           | None -> Lwt.return_ok None
           | Some frame_idx ->
             let key = cache_key_frame page_id frame_idx in
             (match Hashtbl.find_opt t.cache key with
              | Some buf -> Lwt.return_ok (Some (cstruct_dup buf))
              | None ->
                let* r = cb.wal_read_frame frame_idx in
                (match r with
                 | Error s -> Lwt.return_error (Block_error s)
                 | Ok page ->
                   cache_add t key (cstruct_dup page);
                   Lwt.return_ok (Some page)))
       in
       let* wal_r = resolve_via_wal () in
       (match wal_r with
        | Error e -> Lwt.return_error e
        | Ok (Some page) -> Lwt.return_ok page
        | Ok None ->
          let key = cache_key_main page_id in
          (match Hashtbl.find_opt t.cache key with
           | Some buf -> Lwt.return_ok (cstruct_dup buf)
           | None ->
             let buf = Cstruct.create Page.page_size in
             let* result = t.read_page ~page_id buf in
             match result with
             | Error msg -> Lwt.return_error (Block_error msg)
             | Ok () ->
               cache_add t key (cstruct_dup buf);
               Lwt.return_ok buf)))
  | Some max_frame ->
    (* Snapshot reader path: never consult [dirty]. *)
    let resolve_via_wal () =
      match t.wal with
      | None -> Lwt.return_ok None
      | Some cb ->
        match cb.wal_find_page_at page_id ~max_frame with
        | None -> Lwt.return_ok None
        | Some frame_idx ->
          let key = cache_key_frame page_id frame_idx in
          (match Hashtbl.find_opt t.cache key with
           | Some buf -> Lwt.return_ok (Some (cstruct_dup buf))
           | None ->
             let* r = cb.wal_read_frame frame_idx in
             (match r with
              | Error s -> Lwt.return_error (Block_error s)
              | Ok page ->
                cache_add t key (cstruct_dup page);
                Lwt.return_ok (Some page)))
    in
    let* wal_r = resolve_via_wal () in
    (match wal_r with
     | Error e -> Lwt.return_error e
     | Ok (Some page) -> Lwt.return_ok page
     | Ok None ->
       let key = cache_key_main page_id in
       (match Hashtbl.find_opt t.cache key with
        | Some buf -> Lwt.return_ok (cstruct_dup buf)
        | None ->
          let buf = Cstruct.create Page.page_size in
          let* result = t.read_page ~page_id buf in
          match result with
          | Error msg -> Lwt.return_error (Block_error msg)
          | Ok () ->
            cache_add t key (cstruct_dup buf);
            Lwt.return_ok buf))
```

### Step 4.2 — Implement Rwlock

```ocaml
(** Lwt-friendly shared / exclusive lock.

    Many concurrent readers are allowed.  Writers are exclusive of
    other writers AND of readers.  Writer-priority: a pending writer
    prevents new readers from acquiring, preventing writer starvation
    under a steady reader stream.

    Re-entrancy is NOT supported — recursive acquire deadlocks. *)

type t

val create : unit -> t

(** Acquire shared (reader) access for the duration of [f].  Multiple
    concurrent calls run in parallel as long as no writer is active or
    queued. *)
val with_read : t -> (unit -> 'a Lwt.t) -> 'a Lwt.t

(** Acquire exclusive (writer) access for the duration of [f].  Blocks
    new readers and other writers until [f] resolves. *)
val with_write : t -> (unit -> 'a Lwt.t) -> 'a Lwt.t

(** Number of currently-held read locks.  Useful for diagnostics and
    for the checkpoint coordinator to decide if it can proceed. *)
val readers : t -> int

(** True iff a writer is currently holding (or queued for) the lock. *)
val writer_pending : t -> bool
```

- [ ] **Create `lib/store/rwlock.ml`**:

```ocaml
(* Shared/exclusive lock built on Lwt_condition.  Writer-priority. *)

open Lwt.Syntax

type t = {
  mutable readers          : int;
  mutable writer_active    : bool;
  mutable writers_waiting  : int;
  cond                     : unit Lwt_condition.t;
}

let create () =
  { readers = 0;
    writer_active = false;
    writers_waiting = 0;
    cond = Lwt_condition.create () }

let readers t = t.readers
let writer_pending t = t.writer_active || t.writers_waiting > 0

let rec acquire_read t =
  if t.writer_active || t.writers_waiting > 0 then
    let* () = Lwt_condition.wait t.cond in
    acquire_read t
  else begin
    t.readers <- t.readers + 1;
    Lwt.return_unit
  end

let release_read t =
  t.readers <- t.readers - 1;
  if t.readers = 0 then Lwt_condition.broadcast t.cond ()

let rec acquire_write t =
  if t.writer_active || t.readers > 0 then begin
    t.writers_waiting <- t.writers_waiting + 1;
    let* () = Lwt_condition.wait t.cond in
    t.writers_waiting <- t.writers_waiting - 1;
    acquire_write t
  end else begin
    t.writer_active <- true;
    Lwt.return_unit
  end

let release_write t =
  t.writer_active <- false;
  Lwt_condition.broadcast t.cond ()

let with_read t f =
  let* () = acquire_read t in
  Lwt.finalize f (fun () -> release_read t; Lwt.return_unit)

let with_write t f =
  let* () = acquire_write t in
  Lwt.finalize f (fun () -> release_write t; Lwt.return_unit)
```

### Step 6.2 — Checkpoint waits for readers to drain past target frame

The simplest correctness-first model: a checkpoint targets `committed_frames = N` and must wait until `min_active_reader_frames ≥ N` (or no readers). Use an `Lwt_condition` to wake-on-ro_end. Implement narrowly first:

- [ ] **Add** `reader_done_cond : unit Lwt_condition.t` to `bt_state`. Broadcast on `ro_end`.

- [ ] **Rewrite `checkpoint_unlocked`** to:
  1. Capture `target = committed_frames`.
  2. While there exists a reader with snapshot_frames < target: wait on `reader_done_cond`.
  3. Migrate + reset (unchanged).

```ocaml
let rec wait_for_readers_past (st : bt_state) ~target =
  match min_active_reader_frames st with
  | Some m when m < target ->
    let open Lwt.Syntax in
    let* () = Lwt_condition.wait st.reader_done_cond in
    wait_for_readers_past st ~target
  | _ -> Lwt.return_unit
```

- [ ] **Use** in `checkpoint_unlocked` (before iterating the index):

```ocaml
let checkpoint_unlocked (st : bt_state) (wal : Wal.t) : unit Lwt.t =
  let target = Wal.committed_frames wal in
  let* () = wait_for_readers_past st ~target in
  ...
```

(`wait_for_readers_past` uses no lock — it's running under the writer's exclusive section. Readers can still acquire shared because checkpoint already released the write lock if invoked from background; for the inline `PRAGMA wal_checkpoint` path, the writer holds exclusive but new ro_begin calls would block — that's the pre-#149 behaviour and acceptable since explicit checkpoint is rare.)

```ocaml
(** Phase 38 / #149 — WAL reader-scaling benchmark.

    Measures elapsed time for [N] readers × [M] SELECTs vs a baseline
    of 1 reader × N*M SELECTs.  With concurrent readers we expect a
    speedup approaching min(N, host cores). We assert speedup ≥ 3×
    matching the issue's acceptance criterion when [N=8].

    Override via env vars:
      SQLOCAML_BENCH_READERS  (default 8)
      SQLOCAML_BENCH_PER      (default 200)
*)

module Db = Sqlocaml.Db
let run = Lwt_main.run

let getenv_int k d =
  try int_of_string (Sys.getenv k) with _ -> d

let path = "/tmp/sqlocaml_phase38_bench.db"

let setup () =
  (try Unix.unlink path with _ -> ());
  (try Unix.unlink (path ^ "-wal") with _ -> ());
  let db = match run (Db.open_file_wal ~path) with
    | Ok d -> d | Error _ -> failwith "open failed"
  in
  (match run (Db.execute db "CREATE TABLE t (n INTEGER)") with
   | Ok _ -> () | Error _ -> failwith "ddl");
  for i = 0 to 999 do
    match run (Db.execute db (Printf.sprintf "INSERT INTO t VALUES (%d)" i))
    with Ok _ -> () | Error _ -> failwith "seed"
  done;
  db

let bench db ~n_readers ~per =
  let open Lwt.Infix in
  let make_reader _ =
    let rec loop i =
      if i >= per then Lwt.return ()
      else
        Db.query db "SELECT COUNT(*) FROM t" >>= function
        | Error _ -> failwith "select"
        | Ok stream ->
          Lwt_stream.to_list stream >>= fun _ -> loop (i + 1)
    in loop 0
  in
  let t0 = Unix.gettimeofday () in
  run (Lwt.join (List.init n_readers make_reader));
  Unix.gettimeofday () -. t0

let test_scaling () =
  let n_readers = getenv_int "SQLOCAML_BENCH_READERS" 8 in
  let per = getenv_int "SQLOCAML_BENCH_PER" 200 in
  let db = setup () in
  let t_serial = bench db ~n_readers:1 ~per:(n_readers * per) in
  let t_par    = bench db ~n_readers ~per in
  let speedup = t_serial /. t_par in
  Printf.printf "serial=%.3fs parallel=%.3fs speedup=%.2fx\n%!"
    t_serial t_par speedup;
  run (Db.close db);
  (try Unix.unlink path with _ -> ());
  (try Unix.unlink (path ^ "-wal") with _ -> ());
  Alcotest.(check bool) (Printf.sprintf "speedup >= 3x (was %.2fx)" speedup)
    true (speedup >= 3.0)

let () =
  Alcotest.run "wal_reader_scaling" [
    "bench", [
      Alcotest.test_case "8 readers >= 3x serial" `Slow test_scaling;
    ]
  ]
```
