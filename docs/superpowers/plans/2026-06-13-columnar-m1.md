# Columnar Table Type — M1 (In-Memory) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement `CREATE TABLE t (...) USING COLUMNSTORE` as an in-memory columnar table type within sqlocaml, supporting `INSERT INTO col SELECT ...` loading and scan-agg queries via the existing SQL interface.

**Architecture:** A new `lib/columnar/` library provides `Col_store.t` (Bigarray-backed typed column arrays). `Cat.table_meta.storage` becomes a variant (`Row | Columnar of Col_store.t`) so every call site is exhaustiveness-checked by the compiler. Two new plan ops (`Op_col_create_table`, `Op_col_seq_scan`) route columnar DDL and scans through dedicated executor paths; queries still go through the existing `Op_aggregate` for grouping/aggregation.

**Tech Stack:** OCaml 5.4, `Bigarray` (stdlib, MirageOS-compatible), `lwt`, `menhir` (parser), `dune`, `sqlocaml-dev` podman image for all build/test commands.


```ocaml
(* lib/columnar/col.ml *)
open Bigarray
module Row = Sqlocaml_encoding.Row

type int64_arr   = (int64, int64_elt,          c_layout) Array1.t
type float64_arr = (float, float64_elt,         c_layout) Array1.t
type int32_arr   = (int32, int32_elt,           c_layout) Array1.t
type uint8_arr   = (int,   int8_unsigned_elt,   c_layout) Array1.t

type t =
  | I64  of int64_arr
  | F64  of float64_arr
  | Bool of uint8_arr
  | Sym  of { codes : int32_arr; dict : string array }
  | Str  of string array

let length = function
  | I64  a   -> Array1.dim a
  | F64  a   -> Array1.dim a
  | Bool a   -> Array1.dim a
  | Sym  s   -> Array1.dim s.codes
  | Str  a   -> Array.length a

let morsel_size = 1024

let create_for_type (ty : Row.ty) n =
  match ty with
  | Row.Integer -> I64  (Array1.create int64          c_layout n)
  | Row.Real    -> F64  (Array1.create float64         c_layout n)
  | Row.Blob    -> Str  (Array.make n "")
  | Row.Text    ->
    (* Default to Str; Col_store upgrades to Sym if cardinality is low *)
    Str (Array.make n "")

let value_of_col col i =
  match col with
  | I64  a   -> Row.V_int  a.{i}
  | F64  a   -> Row.V_real a.{i}
  | Bool a   -> Row.V_int  (Int64.of_int a.{i})
  | Sym  s   ->
    let code = Int32.to_int s.codes.{i} in
    Row.V_text s.dict.(code)
  | Str  a   -> Row.V_text a.(i)
```

```ocaml
(* lib/columnar/col_store.mli *)
module Row = Sqlocaml_encoding.Row

(** An in-memory columnar table backed by Bigarray columns.
    M1: data is lost on process restart; schema is preserved by the catalog. *)
type t

(** [create columns] allocates an empty store for the given schema.
    Column order matches [columns]. *)
val create : Row.column list -> t

(** Number of rows currently in the store. *)
val nrows : t -> int

(** Column schema this store was created with. *)
val columns : t -> Row.column list

(** Append a batch of rows (each row is an array of values, one per column).
    Values must match the store's column types. Null is stored as a zero/empty sentinel. *)
val insert_rows : t -> Row.t array -> unit

(** Iterate every row as a [Row.t array] in order.
    Morsel batching is internal; callers see a flat sequence. *)
val to_row_seq : t -> Row.t Seq.t
```

```ocaml
(* lib/columnar/col_store.ml *)
open Bigarray
module Row = Sqlocaml_encoding.Row

(* Growable buffer for each column type *)
type buf_col =
  | BI64  of int64          Buffer.t_  (* can't use Bigarray directly as growable *)
  | BF64  of float          Buffer.t_
  | BStr  of string         Queue.t
  | BSym  of { q : string Queue.t; dict : (string, int) Hashtbl.t }

(* We use a simple approach: store all data as OCaml arrays first, then blit to
   Bigarray on first read.  For M1 correctness this is fine; M2 can optimise. *)

type t =
  { schema  : Row.column list
  ; mutable rows : Row.t array list  (* newest batch first *)
  ; mutable total : int
  }

let create schema = { schema; rows = []; total = 0 }

let nrows t = t.total

let columns t = t.schema

let insert_rows t batch =
  if Array.length batch > 0 then begin
    t.rows <- batch :: t.rows;
    t.total <- t.total + Array.length batch
  end

let to_row_seq t =
  (* Reverse to get insertion order, then flatten *)
  let all = List.rev t.rows in
  let batches = Array.of_list all in
  let bi = ref 0 and ri = ref 0 in
  let rec next () =
    if !bi >= Array.length batches then Seq.Nil
    else
      let batch = batches.(!bi) in
      if !ri >= Array.length batch then begin
        incr bi; ri := 0; next ()
      end else begin
        let row = batch.(!ri) in
        incr ri;
        Seq.Cons (row, next)
      end
  in
  next
```

Replace the two functions (lines 535–566) with:

```ocaml
(* storage_kind trailing byte: 0 = Row (default; absent in old encodings), 1 = Columnar *)
let encode_table_value m =
  let buf = Buffer.create 16 in
  (match m.storage with
   | Row { tree_id; next_rowid; without_rowid; autoincrement } ->
     Varint.encode_uint64 buf (Int64.of_int tree_id);
     Varint.encode_int64  buf next_rowid;
     Varint.encode_uint64 buf (if without_rowid  then 1L else 0L);
     Varint.encode_uint64 buf (if autoincrement  then 1L else 0L);
     Varint.encode_uint64 buf 0L  (* storage_kind = Row *)
   | Columnar _ ->
     Varint.encode_uint64 buf 0L;  (* dummy tree_id *)
     Varint.encode_int64  buf empty_next_rowid;
     Varint.encode_uint64 buf 0L;  (* without_rowid = false *)
     Varint.encode_uint64 buf 0L;  (* autoincrement = false *)
     Varint.encode_uint64 buf 1L   (* storage_kind = Columnar *)
  );
  Buffer.to_bytes buf

(* Returns (storage, columns_needed_for_col_store).
   Columns are not decoded here; caller passes them when constructing Columnar. *)
let decode_table_storage bytes columns =
  let tid,  off   = Varint.decode_uint64 bytes 0 in
  let next, off'  = Varint.decode_int64  bytes off in
  let without_rowid, off'' =
    if off' >= Bytes.length bytes then false, off'
    else (let v, o = Varint.decode_uint64 bytes off' in Int64.to_int v <> 0, o)
  in
  let autoincrement, off''' =
    if off'' >= Bytes.length bytes then false, off''
    else (let v, o = Varint.decode_uint64 bytes off'' in Int64.to_int v <> 0, o)
  in
  let storage_kind =
    if off''' >= Bytes.length bytes then 0
    else (let v, _ = Varint.decode_uint64 bytes off''' in Int64.to_int v)
  in
  match storage_kind with
  | 1 -> Columnar (Sqlocaml_columnar.Col_store.create columns)
  | _ -> Row { tree_id = Int64.to_int tid; next_rowid = next; without_rowid; autoincrement }
```

```ocaml
let create_columnstore_table ?txn t ~name ~columns =
  if Schema_cache.mem_table t.sc name
  then failwith (Printf.sprintf "table '%s' already exists" name);
  let col_store = Sqlocaml_columnar.Col_store.create columns in
  let m =
    { name
    ; storage = Columnar col_store
    ; columns
    ; fk_constraints = []
    }
  in
  let write_rows tx =
    (* Write to sys_tables using encode_table_value (Columnar branch writes storage_kind=1) *)
    let%lwt () = S.put tx sys_tables_tid (Bytes.of_string name) (encode_table_value m) in
    Lwt_list.iteri_s
      (fun i col -> S.put tx sys_columns_tid (column_key name i) (encode_column col))
      columns
  in
  match txn with
  | Some tx ->
    let%lwt () = write_rows tx in
    Schema_cache.put_table t.sc ~name m;
    Lwt.return ()
  | None ->
    let%lwt tx = S.rw_begin t.store in
    let%lwt () = write_rows tx in
    let%lwt () = S.commit tx in
    Schema_cache.put_table_durable t.sc ~name m;
    Lwt.return ()
```

```ocaml
let test_schema_persists_data_empty () =
  (* In-memory db: re-open loses data but schema is preserved via catalog *)
  (* This test uses file-backed db to verify schema reload *)
  let path = Filename.temp_file "sqlocaml_col_" ".db" in
  let* db = Db.open_ ~path:(Some path) () in
  let* () = Db.exec db "CREATE TABLE t (x REAL) USING COLUMNSTORE" in
  let* () = Db.exec db "INSERT INTO t VALUES (1.0), (2.0)" in
  let* () = Db.close db in
  let* db2 = Db.open_ ~path:(Some path) () in
  (* Table should exist (schema preserved) *)
  let* rows = Db.query db2 "SELECT name FROM sqlite_master WHERE type='table'" in
  let names = List.filter_map (function
    | [| Db.V_text n |] -> Some n | _ -> None) rows in
  assert (List.mem "t" names);
  (* Data is gone (M1 limitation) *)
  let* rows2 = Db.query db2 "SELECT COUNT(*) FROM t" in

  let* () = Db.close db2 in
  (try Unix.unlink path with _ -> ());
```
