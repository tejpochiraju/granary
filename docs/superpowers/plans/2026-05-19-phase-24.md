# Phase 24 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement three independent SQL features: case-insensitive keywords (#112), LIMIT/OFFSET on UPDATE/DELETE (#121), and CURRENT_TIMESTAMP/DATE/TIME default expressions (#59).

**Architecture:** (1) Lexer refactor routes all keyword recognition through the `ident` rule via `String.uppercase_ascii`, making every SQL keyword case-insensitive. (2) UPDATE/DELETE LIMIT/OFFSET threads new fields through the AST→Sema→Plan→Exec pipeline and slices the row list inside `execute_update`/`execute_delete` before writing. (3) Default expressions add three new `Row.default_value` variants (`DV_current_timestamp`, `DV_current_date`, `DV_current_time`) and three matching `Ast.literal` variants; sema converts them to `BE_func` so they are evaluated at INSERT time rather than at parse time.

**Tech Stack:** OCaml 5.x, dune 3.x, menhir (parser.mly), ocamllex (lexer.mll), lwt, alcotest. All dune commands run inside podman: `podman run --rm -v $(pwd):/workspace:Z -w /workspace sqlocaml-dev dune <cmd>`. Tests are in `test/test_e2e.ml` and `test/test_sqlite_compare.ml`.

let bind_update cat ~param_counter ~named_params ~table ~assignments ~where ~order ~limit ~offset ~returning =
  let* meta_opt = Cat.find_table cat ~name:table in
  match meta_opt with
  | None -> Lwt.return (Error (Unknown_table table))
  | Some meta ->
    let bind_assign_result =
      List.fold_left (fun acc (col_name, rhs_expr) ->
        match acc with
        | Error _ -> acc
        | Ok bound_list ->
          (match col_index meta.columns col_name with
           | None   -> Error (Unknown_column { table; column = col_name })
           | Some i ->
             (match bind_expr ~param_counter ~named_params meta rhs_expr with
              | Error e -> Error e
              | Ok be   -> Ok (bound_list @ [(i, be)])))
      ) (Ok []) assignments
    in
    (match bind_assign_result with
     | Error e -> Lwt.return (Error e)
     | Ok bound_assigns ->
       let where_result =
         match where with
         | None   -> Ok None
         | Some e ->
           (match bind_expr ~param_counter ~named_params meta e with
            | Ok be   -> Ok (Some be)
            | Error e -> Error e)
       in
       (match where_result with
        | Error e -> Lwt.return (Error e)
        | Ok bound_where ->
          let has_subquery_in_where = match bound_where with
            | Some e -> expr_has_subquery e
            | None   -> false
          in
          let has_subquery_in_assign =
            List.exists (fun (_, e) -> expr_has_subquery e) bound_assigns
          in
          if has_subquery_in_where || has_subquery_in_assign then
            Lwt.return (Error (Unsupported "subqueries in UPDATE WHERE/SET are not supported"))
