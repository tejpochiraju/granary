
# Phase 28: Quoted Identifier Escaping, EXPLAIN/EXPLAIN ANALYZE, CLI

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship three SQL-compat gap closures: (1) escaped quoted identifiers (`"foo""bar"`, `` `foo``bar` ``, `[foo]bar]`); (2) `EXPLAIN` / `EXPLAIN ANALYZE` query plan visualization; (3) `sqlocaml_cli` binary for executing SQL files.

**Architecture:** Quoted ident fix is lexer-only (ocamllex auxiliary rules). EXPLAIN threads through the full pipeline (AST → Sema → Plan → Exec) as a wrapper that walks the plan tree and emits rows; EXPLAIN ANALYZE additionally executes the inner query and records row count + elapsed time. The CLI is a new dune executable that opens a database, reads SQL, routes each statement to `Db.query` or `Db.execute_change_count`, and prints pipe-delimited results.

**Tech Stack:** OCaml 5.x, dune 3.x, lwt, menhir, alcotest. Build command: `podman run --rm -v $(pwd):/workspace:Z -w /workspace sqlocaml-dev dune <cmd>`.  Test command: same with `dune test`. All dune commands MUST run inside podman — never call dune directly on the host.

- [ ] **Step 3: Replace the three simple quoted-ident rules with auxiliary rules**

In `lib/sql/lexer.mll`, locate lines 45–47:
```ocaml
  | '"' ([^ '"']* as id) '"'  { IDENT id }
  | '`' ([^ '`']* as id) '`'  { IDENT id }
  | '[' ([^ ']']* as id) ']'  { IDENT id }
```

Replace them with:
```ocaml
  | '"'  { read_double_quoted (Buffer.create 16) lexbuf }
  | '`'  { read_backtick_quoted (Buffer.create 16) lexbuf }
  | '['  { read_bracket_quoted (Buffer.create 16) lexbuf }
```

Then, at the very end of the file (after the closing `}` of the `token` rule), add the three auxiliary rules:

```ocaml
and read_double_quoted buf = parse
  | '"' '"'    { Buffer.add_char buf '"'; read_double_quoted buf lexbuf }
  | '"'        { IDENT (Buffer.contents buf) }
  | [^ '"']+   { Buffer.add_string buf (Lexing.lexeme lexbuf);
                 read_double_quoted buf lexbuf }
  | eof        { failwith "unterminated quoted identifier" }

and read_backtick_quoted buf = parse
  | '`' '`'   { Buffer.add_char buf '`'; read_backtick_quoted buf lexbuf }
  | '`'       { IDENT (Buffer.contents buf) }
  | [^ '`']+  { Buffer.add_string buf (Lexing.lexeme lexbuf);
                read_backtick_quoted buf lexbuf }
  | eof       { failwith "unterminated quoted identifier" }

and read_bracket_quoted buf = parse
  | "]]"      { Buffer.add_char buf ']'; read_bracket_quoted buf lexbuf }
  | ']'       { IDENT (Buffer.contents buf) }
  | [^ ']']+  { Buffer.add_string buf (Lexing.lexeme lexbuf);
                read_bracket_quoted buf lexbuf }
  | eof       { failwith "unterminated quoted identifier" }
```

**Important:** ocamllex auxiliary rules use `and rulename args = parse` syntax. The `buf` argument is passed explicitly — this is valid ocamllex. The rules must appear after the main `token` rule.

```ocaml
    Each row has 3 elements: [id (INT), parent_id (INT), detail (TEXT)].
    Root node has parent_id = -1. *)

  let counter = ref 0 in
  let rec walk parent op =
    let id = !counter in
    incr counter;
    let my_row = [| Row.V_int (Int64.of_int id);
                    Row.V_int (Int64.of_int parent);
                    Row.V_text (op_name op) |] in
```

```ocaml
open Lwt.Syntax
module Db  = Sqlocaml.Db

let print_value ppf = function
  | Db.V_null   -> Format.pp_print_string ppf "NULL"
  | Db.V_int n  -> Format.pp_print_string ppf (Int64.to_string n)
  | Db.V_real f -> Format.fprintf ppf "%.17g" f
  | Db.V_text s -> Format.pp_print_string ppf s
  | Db.V_blob b -> Format.fprintf ppf "<blob:%d>" (Bytes.length b)

let print_row row =
  Array.iteri (fun i v ->
    if i > 0 then print_char '|';
    print_value Format.std_formatter v
  ) row;
  print_newline ()

(** Guess whether a statement should be routed to Db.query (returns rows)
    or Db.execute_change_count (returns affected count).
    Uses the first keyword of the trimmed statement. *)
let is_query_stmt sql =
  let s = String.trim sql in
  if s = "" then false
  else
    let upper = String.uppercase_ascii s in
    let first_word =
      match String.index_opt upper ' ', String.index_opt upper '\n',
            String.index_opt upper '\t' with
      | None,   None,   None   -> upper
      | Some i, None,   None   -> String.sub upper 0 i
      | None,   Some i, None   -> String.sub upper 0 i
      | None,   None,   Some i -> String.sub upper 0 i
      | Some a, Some b, None   -> String.sub upper 0 (min a b)
      | Some a, None,   Some b -> String.sub upper 0 (min a b)
      | None,   Some a, Some b -> String.sub upper 0 (min a b)
      | Some a, Some b, Some c -> String.sub upper 0 (min a (min b c))
    in
    match first_word with
    | "SELECT" | "WITH" | "EXPLAIN" | "VALUES" | "PRAGMA" -> true
    | _ -> false

let run_sql db sql =
  let stmts =
    String.split_on_char ';' sql
    |> List.map String.trim
    |> List.filter (fun s -> s <> "")
  in
  Lwt_list.iter_s (fun stmt ->
    if is_query_stmt stmt then begin
      let* result = Db.query db stmt in
      match result with
      | Error e ->
        Printf.eprintf "Error: %a\n%!" Db.pp_error e;
        Lwt.return_unit
      | Ok stream ->
        Lwt_stream.iter print_row stream
    end else begin
      let* result = Db.execute_change_count db stmt in
      match result with
      | Error e ->
        Printf.eprintf "Error: %a\n%!" Db.pp_error e;
        Lwt.return_unit
      | Ok n ->
        if n > 0 then Printf.printf "%d row(s) affected\n%!" n;
        Lwt.return_unit
    end
  ) stmts

let read_file path =
  let ic = open_in path in
  let n = in_channel_length ic in
  let s = Bytes.create n in
  really_input ic s 0 n;
  close_in ic;
  Bytes.to_string s

let read_stdin () =
  let buf = Buffer.create 256 in
  (try while true do
    Buffer.add_channel buf stdin 4096
  done with End_of_file -> ());
  Buffer.contents buf

let () =
  let args = Array.to_list Sys.argv |> List.tl in
  let (db_path, sql_source) = match args with
    | []             -> (":memory:", `Stdin)
    | [path]         -> (path, `Stdin)
    | path :: f :: _ -> (path, `File f)
  in
  let sql = match sql_source with
    | `Stdin    -> read_stdin ()
    | `File f   -> read_file f
  in
  Lwt_main.run (
    let* db =
      if db_path = ":memory:" then
        Db.open_in_memory ()
      else begin
        let* result = Db.open_file ~path:db_path in
        match result with
        | Ok db -> Lwt.return db
        | Error e ->
          Printf.eprintf "Cannot open '%s': %a\n%!" db_path Db.pp_error e;
          exit 1
      end
    in
    let* () = run_sql db sql in
    Db.close db
  )
```
